name: Deploy static site

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Verify solver against all 1820 hands
        run: node tests/engine.test.js

  deploy:
    needs: test
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.out.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4

      - name: Assemble site
        run: |
          mkdir -p _site/js
          cp index.html styles.css _site/
          cp js/engine.js js/data.js js/ui.js _site/js/

      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: _site
      - id: out
        uses: actions/deploy-pages@v4
